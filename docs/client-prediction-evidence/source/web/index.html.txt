    <script type="module">
      import init from "./ryggattack.js";
      import { calibrateRefreshRate } from "./refresh-rate.mjs";
      import { installRoomControls } from "./rooms.mjs";
      import { installOnlineHud } from "./online-hud-bridge.mjs";
      installRoomControls();
      installOnlineHud();

      const status = document.getElementById("status");
      const canvas = document.getElementById("ryggattack-canvas");

      // Winit listens for key events on the canvas, so it has to hold focus
      // for the keyboard controls to work.
      if (document.getElementById('online-entry').hidden) canvas.focus();
      window.addEventListener("pointerdown", (event) => {
        if (!event.target.closest("#online-entry, .rygg-online-hud")) canvas.focus();
      });

      try {
        // No game/GPU work during calibration: slow gameplay must never be
        // mistaken for a low-refresh display. Rust validates this hint too.
        window.ryggattackRefreshRate = await calibrateRefreshRate().catch(() => null);
        await init();
        status.remove();
      } catch (error) {
        // Winit stops its web event loop by throwing, so this particular
        // exception means the game started rather than that it failed.
        if (String(error.message).startsWith("Using exceptions for control flow")) {
          status.remove();
        } else {
          status.classList.add("failed");
          status.textContent = `Ryggattack failed to start: ${error}`;
          throw error;
        }
      }
    </script>
