import crypto from 'node:crypto';
import zlib from 'node:zlib';

const sha=b=>crypto.createHash('sha256').update(b).digest('hex');
const copy=(from,to)=>{fs.mkdirSync(path.dirname(to),{recursive:true});fs.copyFileSync(from,to);};
const off=path.join(repo,'verification/parity/paired-offline-slow-q0'), offNormal=path.join(repo,'verification/parity/paired-offline-wide-q0');

const offVideo=path.join(off,offReport.video),onVideo=path.join(on,'private-raw-video/StageHuman.webm');
const provenance={created_at:new Date().toISOString(),scope:'Real rendered matched quality0 and960x480; slow visual fixtures supplement normal-speed gameplay, not timing/network acceptance',viewport:[960,480],quality:0,clips:[],frames:[],private_raw_videos_not_published:true};

  const bytes=fs.readFileSync(path.join(dir,f+'.json'));fs.writeFileSync(path.join(base,name,f+'.json.gz'),zlib.gzipSync(bytes));
 }
}
function clip(name,source,start,duration){
 const file=`${name}.webm`;ff(['-ss',String(start),'-i',source,'-t',String(duration),'-an','-c:v','libvpx-vp9','-threads','1','-crf','34','-b:v','0',path.join(base,file)]);
 provenance.clips.push({file,source_sha256:sha(fs.readFileSync(source)),source_start_seconds:start,duration_seconds:duration,operation:'ffmpeg trim/re-encode only; original timestamps and source SHA retained; no invented/interpolated frames'});
}
clip('offline-slow-sequence',offVideo,65,10);
clip('online-slow-sequence',onVideo,18,10);
clip('offline-normal-sequence',path.join(offNormal,offNormalReport.video),122,16);

 const file=`${kind}-${stage}.png`;ff(['-i',source,'-vf',`select=gte(t\\,${t})`,'-frames:v','1',path.join(base,file)]);provenance.frames.push({file,source_sha256:sha(fs.readFileSync(source)),source_seconds:t,stage});
}
for(const [kind,source,t] of [['offline',offVideo,67.2],['online',onVideo,19.68]]){

}
// Annotated montage preserves full framing. Detail crops use identical unscaled source rectangle.
for(const f of provenance.frames){const temp=path.join(base,'.montage-'+f.file);const label=f.file.replace('.png','')+' source '+f.source_seconds.toFixed(1)+'s';ff(['-i',path.join(base,f.file),'-vf',`scale=480:240,pad=480:264:0:24:color=0x101820,drawtext=text='${label}':x=8:y=4:fontsize=14:fontcolor=white`,'-frames:v','1',temp]);}
const montageInputs=provenance.frames.map(f=>path.join(base,'.montage-'+f.file));
// use concat demuxer for still frames then tile in deterministic stage order
fs.writeFileSync(path.join(base,'.montage-list'),montageInputs.map(f=>`file '${f}'`).join('\n'));

for(const f of provenance.frames){ff(['-i',path.join(base,f.file),'-vf',`crop=200:170:590:40,pad=240:200:0:30:color=0x101820,drawtext=text='${f.file.replace('.png','')}':x=4:y=5:fontsize=11:fontcolor=white`,'-frames:v','1',path.join(base,'.detail-'+f.file)]);}
fs.writeFileSync(path.join(base,'.detail-list'),provenance.frames.map(f=>`file '${path.join(base,'.detail-'+f.file)}'`).join('\n'));
ff(['-f','concat','-safe','0','-i',path.join(base,'.detail-list'),'-vf','tile=2x5','-frames:v','1',path.join(base,'paired-detail.png')]);
for(const entry of fs.readdirSync(base))if(entry.startsWith('.'))fs.rmSync(path.join(base,entry));
